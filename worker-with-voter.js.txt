export default {
  async fetch(request, env) {
    // শুধু তোমার সাইট থেকেই কাজ করবে - অন্য সার্ভার ব্লক
    const allowedOrigin = "https://kalimata352-blip.github.io";
    const origin = request.headers.get("Origin") || "";
    // GitHub pages + localhost allow (testing এর জন্য)
    const isAllowed = origin.includes("kalimata352-blip.github.io") || origin.includes("localhost") || !origin;

    const cors = {
      "Access-Control-Allow-Origin": isAllowed ? (origin || allowedOrigin) : allowedOrigin,
      "Access-Control-Allow-Methods": "GET, POST, DELETE, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type, x-shared-token",
      "Vary": "Origin"
    };

    if (request.method === "OPTIONS") return new Response(null, { headers: cors });

    // অন্য সার্ভার থেকে POST/DELETE ব্লক
    if (!isAllowed && request.method !== "GET") {
      return new Response(JSON.stringify({ error: "Unauthorized origin - অন্য সার্ভার থেকে কাজ করবে না" }), { status: 403, headers: { ...cors, "Content-Type": "application/json" } });
    }

    // টেবিল তৈরি
    await env.DB.prepare(`CREATE TABLE IF NOT EXISTS products (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, price REAL NOT NULL, image TEXT)`).run();
    await env.DB.prepare(`CREATE TABLE IF NOT EXISTS staff (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      phone TEXT NOT NULL,
      role TEXT,
      voter_card TEXT NOT NULL UNIQUE,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )`).run();

    const url = new URL(request.url);

    // ========== PRODUCTS API ==========
    if (url.pathname === "/api/products") {
      const h = { ...cors, "Content-Type": "application/json" };
      if (request.method === "GET") {
        const { results } = await env.DB.prepare("SELECT * FROM products ORDER BY id DESC").all();
        return new Response(JSON.stringify(results), { headers: h });
      }
      if (request.method === "POST") {
        const { name, price, image } = await request.json();
        if (!name || !price) return new Response(JSON.stringify({ error: "name price required" }), { status: 400, headers: h });
        await env.DB.prepare("INSERT INTO products (name, price, image) VALUES (?,?,?)").bind(name, price, image || "").run();
        return new Response(JSON.stringify({ success: true }), { headers: h });
      }
      if (request.method === "DELETE") {
        const id = url.searchParams.get("id");
        await env.DB.prepare("DELETE FROM products WHERE id=?").bind(id).run();
        return new Response(JSON.stringify({ success: true }), { headers: h });
      }
    }

    // ========== KORMI API - ভোটার কার্ড সহ ==========
    if (url.pathname === "/api/kormi") {
      const h = { ...cors, "Content-Type": "application/json" };

      if (request.method === "GET") {
        const { results } = await env.DB.prepare("SELECT * FROM staff ORDER BY id DESC").all();
        return new Response(JSON.stringify(results), { headers: h });
      }

      if (request.method === "POST") {
        const body = await request.json();
        let { name, phone, role, voter_card } = body;

        // ভোটার কার্ড মাস্ট
        if (!name || !phone || !voter_card) {
          return new Response(JSON.stringify({ error: "নাম, ফোন ও ভোটার কার্ড বাধ্যতামূলক" }), { status: 400, headers: h });
        }

        voter_card = voter_card.trim().toUpperCase().replace(/\s/g, "");

        // ভোটার কার্ড ফরম্যাট চেক (ABC1234567 এর মতো)
        if (voter_card.length < 6) {
          return new Response(JSON.stringify({ error: "ভোটার কার্ড সঠিক নয়" }), { status: 400, headers: h });
        }

        // একই ভোটার কার্ড আছে কিনা চেক - ডুপ্লিকেট ব্লক
        const existing = await env.DB.prepare("SELECT id FROM staff WHERE voter_card = ?").bind(voter_card).first();
        if (existing) {
          return new Response(JSON.stringify({ error: "এই ভোটার কার্ড দিয়ে আগেই আবেদন করা হয়েছে! একই ভোটার কার্ড বার বার চলবে না।" }), { status: 409, headers: h });
        }

        try {
          await env.DB.prepare("INSERT INTO staff (name, phone, role, voter_card) VALUES (?,?,?,?)").bind(name, phone, role || "", voter_card).run();
          return new Response(JSON.stringify({ success: true, message: "কর্মী যোগ হয়েছে" }), { headers: h });
        } catch (e) {
          // UNIQUE constraint error
          if (e.message && e.message.includes("UNIQUE")) {
            return new Response(JSON.stringify({ error: "এই ভোটার কার্ড আগে ব্যবহার হয়েছে" }), { status: 409, headers: h });
          }
          return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: h });
        }
      }

      if (request.method === "DELETE") {
        const id = url.searchParams.get("id");
        await env.DB.prepare("DELETE FROM staff WHERE id=?").bind(id).run();
        return new Response(JSON.stringify({ success: true }), { headers: h });
      }
    }

    return new Response(JSON.stringify({ ok: true, msg: "API Running - use /api/products and /api/kormi" }), { headers: { ...cors, "Content-Type": "application/json" } });
  }
}
